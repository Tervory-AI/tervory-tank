import os, pathlib, socket, subprocess, sys
import pgserver
from pgserver.postgres_server import pg_ctl
hints=('API_KEY','TOKEN','SECRET','PASSWORD','DATABASE_URL','OPENAI','ANTHROPIC','OPENROUTER','PERPLEXITY','DEEPSEEK','GROQ','TOGETHER','MISTRAL','GEMINI','XAI_','HF_')
env={k:v for k,v in os.environ.items() if not any(h in k.upper() for h in hints)}
env['PYTHONPATH']='tank:agents/_chassis'
data=pathlib.Path('.platform-proof/tmp/placeholder_pg').resolve()
server=pgserver.get_server(data)
try:
    pg_ctl(['-m','fast','-w','stop'],pgdata=data,user=server.system_user,timeout=15)
    with socket.socket() as handle:
        handle.bind(('127.0.0.1',0))
        port=handle.getsockname()[1]
    pg_ctl(['-w','-o',f'-h 127.0.0.1 -p {port} -k {data}','-l',str(data/'log'),'start'],pgdata=data,user=server.system_user,timeout=15)
    env['POOL_SCHEMA_DATABASE_URL']=f'postgresql://postgres@127.0.0.1:{port}/postgres'
    env['TMPDIR']='/workspace/codex-platform-w3/.platform-proof/tmp'
    print('Disposable PostgreSQL started in the assigned checkout; external sockets disabled.',flush=True)
    result=subprocess.call([sys.executable,'-m','pytest','-q','--disable-socket','--allow-unix-socket','--allow-hosts=127.0.0.1,localhost,::1',*sys.argv[1:]],env=env)
finally:
    server.cleanup()
raise SystemExit(result)
